import pandas as pd
import plotly.express as px

# Đọc dữ liệu từ file CSV
df = pd.read_csv("data/dataset.csv")

# Chuyển đổi cột 'date' sang định dạng datetime
df['date'] = pd.to_datetime(df['date'])

# Sắp xếp dữ liệu theo ngày để đảm bảo biểu đồ hiển thị đúng xu hướng thời gian
df = df.sort_values(by='date')

# Tính toán nhiệt độ trung bình cao nhất và thấp nhất trong toàn bộ dataset
overall_max_temp = df['temperature_mean'].max()
overall_min_temp = df['temperature_mean'].min()

# In ra giá trị max/min
print(f"Nhiệt độ trung bình cao nhất trong 7 ngày qua: {overall_max_temp:.1f} °C")
print(f"Nhiệt độ trung bình thấp nhất trong 7 ngày qua: {overall_min_temp:.1f} °C")

# Tạo biểu đồ đường (line chart) với màu sắc theo tỉnh và trendline OLS cho từng tỉnh
fig = px.line(df,
              x='date',
              y='temperature_mean',
              color='province',
              title='Xu hướng nhiệt độ trung bình 7 ngày gần nhất theo tỉnh',
              labels={'date': 'Ngày', 'temperature_mean': 'Nhiệt độ trung bình (°C)'},
              trendline='ols') # Thêm đường xu hướng OLS cho mỗi tỉnh

# Cập nhật layout để đảm bảo nhãn trục rõ ràng
fig.update_layout(xaxis_title='Ngày', yaxis_title='Nhiệt độ trung bình (°C)')
